* Defaults

Sane behavior that isn't worth a package.

#+begin_src emacs-lisp
;; -*- lexical-binding: t; -*-
(setq inhibit-startup-screen t)
(setq ring-bell-function 'ignore)
(setq use-short-answers t)

(global-auto-revert-mode 1)
(save-place-mode 1)
(show-paren-mode 1)

;; Keep backup/autosave files out of the working directory.
(let ((backup-dir (expand-file-name "backups/" user-emacs-directory)))
  (make-directory backup-dir t)
  (setq backup-directory-alist `((".*" . ,backup-dir))
        auto-save-file-name-transforms `((".*" ,backup-dir t))))
#+end_src

* Editing this config

Quick access to the literate config itself.

#+begin_src emacs-lisp
(defun my/find-config ()
  "Open this literate config file for editing."
  (interactive)
  (find-file (expand-file-name "config.org" user-emacs-directory)))

(global-set-key (kbd "C-c e") #'my/find-config)
#+end_src

* Completion

Vertico for the minibuffer, orderless for fuzzy matching, marginalia for
annotations.

#+begin_src emacs-lisp
(use-package vertico
  :init (vertico-mode))

(use-package orderless
  :custom (completion-styles '(orderless basic)))

(use-package marginalia
  :init (marginalia-mode))
#+end_src
